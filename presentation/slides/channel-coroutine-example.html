Channels with Coroutines


async::awaitable<void> producer(async::experimental::channel<void(error_code, int)>& ch) {
    for (int counter = 0; ; ++counter) {
        co_await ch.async_send(error_code{}, counter, async::use_awaitable);
    }
}

async::awaitable<void> consumer(async::experimental::channel<void(error_code, int)>& ch) {
    for (;;) {
        auto [ec, value] = co_await ch.async_receive(async::use_awaitable);
        if (!ec) {
            std::cout << "Received: " << value << std::endl;
        }
    }
}

int main() {
    async::io_context io;
    async::experimental::channel<void(error_code, int)> channel(io, 10);

    async::co_spawn(io, producer(channel), async::detached);
    async::co_spawn(io, consumer(channel), async::detached);

    io.run();
}

Channels: Producer-Consumer


void producer(async::experimental::channel<void(error_code, int)>& ch) {
    static int counter = 0;
    ch.async_send(
        error_code{}, counter,
        [&ch](error_code ec) {
            if (!ec) {
                ++counter;
                producer(ch);
            }
        });
}

void consumer(async::experimental::channel<void(error_code, int)>& ch) {
    ch.async_receive(
        [&ch](error_code ec, int value) {
            if (!ec) {
                std::cout << "Received: " << value << std::endl;
                consumer(ch);
            }
        });
}

int main() {
    async::io_context io;
    async::experimental::channel<void(error_code, int)> channel(io, 10);

    producer(channel);
    consumer(channel);

    io.run();
}

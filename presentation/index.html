Boost.Asio

Asynchronous programming in Modern C++

High Performance • Scalable • Cross‑Platform

What is Boost.Asio?

  • Library for network & low‑level I/O
  • Asynchronous event-driven programming
  • Part of the Boost C++ Libraries
  • Non-Boost version available
  • Forms the base for std::executors in Networking TS

Key Features

  • TCP / UDP Networking
  • Timers & Serial Ports
  • Strand-based concurrency control
  • Supports callbacks, futures, coroutines
  • Built-in TLS support
  • Channels

Core Concept: io_context

Event processing core of Asio applications


            boost::asio::io_context io;

            io.run();  // processes all async operations

Simple Example: Hello World


#include <iostream>
#include <boost/asio.hpp>

namespace async = boost::asio;

int main() {
    async::io_context io;

    async::post(io, []() {
        std::cout << "Hello World" << std::endl;
    });

    io.run();
}

Output:

$ ./main
Hello World

Timer Loop Example


void async_function(async::steady_timer* timer) {
    static auto counter = 0;
    timer->expires_after(std::chrono::seconds(1));
    timer->async_wait([timer](const boost::system::error_code&) {
        std::cout << "Hello World: " << counter << std::endl;
        ++counter;
        async_function(timer);
    });
}

int main() {
    async::io_context io;
    async::steady_timer t(io);
    async_function(&t);
    io.run();
}

Output (1 second intervals):

$ ./timer_loop_example

Channels: Producer-Consumer


void producer(async::experimental::channel<void(error_code, int)>& ch) {
    static int counter = 0;
    ch.async_send(
        error_code{}, counter,
        [&ch](error_code ec) {
            if (!ec) {
                ++counter;
                producer(ch);
            }
        });
}

void consumer(async::experimental::channel<void(error_code, int)>& ch) {
    ch.async_receive(
        [&ch](error_code ec, int value) {
            if (!ec) {
                std::cout << "Received: " << value << std::endl;
                consumer(ch);
            }
        });
}

int main() {
    async::io_context io;
    async::experimental::channel<void(error_code, int)> channel(io, 10);

    producer(channel);
    consumer(channel);

    io.run();
}

Output:

$ ./channel_example
Received: 0
Received: 1
Received: 2
Received: 3
Received: 4
...

Channels with Coroutines


async::awaitable<void> producer(async::experimental::channel<void(error_code, int)>& ch) {
    for (int counter = 0; ; ++counter) {
        co_await ch.async_send(error_code{}, counter, async::use_awaitable);
    }
}

async::awaitable<void> consumer(async::experimental::channel<void(error_code, int)>& ch) {
    for (;;) {
        auto [ec, value] = co_await ch.async_receive(async::use_awaitable);
        if (!ec) {
            std::cout << "Received: " << value << std::endl;
        }
    }
}

int main() {
    async::io_context io;
    async::experimental::channel<void(error_code, int)> channel(io, 10);

    async::co_spawn(io, producer(channel), async::detached);
    async::co_spawn(io, consumer(channel), async::detached);

    io.run();
}

Output:

$ ./channel_coroutine_example

Received: 4
